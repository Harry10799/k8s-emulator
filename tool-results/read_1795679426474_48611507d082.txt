   190→export function parseSchedule(schedule: string): number {
   191→  const m = schedule.match(/^\*\/(\d+)/)
   192→  if (m) return Math.max(1, Number(m[1]))
   193→  if (schedule.trim() === '* * * * *') return 60
   194→  const plain = schedule.match(/^(\d+)$/)
   195→  if (plain) return Math.max(1, Number(plain[1]))
   196→  return 0 // unsupported
   197→}

   262→function logEvent(state: Pick<ClusterState, 'events' | 'tickCount'>, type: 'Normal' | 'Warning', reason: string, message: string, object: string) {
   263→  state.events.unshift({ id: uid('ev'), time: simTime(state.tickCount), type, reason, message, object })
   264→  if (state.events.length > 100) state.events.length = 100
   265→}

   267→function podLog(pod: Pod, tick: number, line: string) {
   268→  pod.logs.push(`[${simTime(tick)}] ${line}`)
   269→  if (pod.logs.length > 30) pod.logs.splice(0, pod.logs.length - 30)
   270→}

   272→export function nodeUsage(pods: Pod[], nodeId: string) {
   273→  return pods
   274→    .filter((p) => p.requiredNode === nodeId || (p.requiredNode === null && p.nodeId === nodeId))
   275→    .filter((p) => p.phase !== 'Terminating' && p.phase !== 'Succeeded')
   276→    .reduce(
   277→      (acc, p) => {
   278→        acc.cpu += p.cpuRequest
   279→        acc.mem += p.memRequest
   280→        return acc
   281→      },
   282→      { cpu: 0, mem: 0 }
   283→    )
   284→}

   315→export const useCluster = create<ClusterState>((set, get) => ({
   316→  nodes: [
   317→    { id: 'cp1', name: 'control-plane-1', role: 'control-plane', cpuCapacity: 4000, memCapacity: 8192, position: { x: 300, y: 0 } },
   318→    { id: 'n1', name: 'worker-node-1', role: 'worker', cpuCapacity: 4000, memCapacity: 8192, position: { x: 0, y: 180 } },
   319→    { id: 'n2', name: 'worker-node-2', role: 'worker', cpuCapacity: 4000, memCapacity: 8192, position: { x: 600, y: 180 } },
   320→  ],
   321→  pods: [],
   322→  deployments: [],
   323→  replicaSets: [],
   324→  daemonSets: [],
   325→  statefulSets: [],
   326→  services: [],
   327→  configMaps: [],
   328→  secrets: [],
   329→  cronJobs: [],
   330→  jobs: [],
   331→  namespaces: [{ id: 'ns-default', name: 'default', labels: { 'kubernetes.io/metadata.name': 'default' } }],
   332→  activeNamespace: 'all',
   333→  internalsNode: null,
   334→  events: [],
   335→  tickCount: 0,
   336→  running: true,
   337→  speed: 1,
   338→  chaos: false,
   339→  selected: null,
   340→  editing: null,
   341→
   342→  tick: () =>
   343→    set((state) => {
   344→      const s: ClusterState = {
   345→        ...state,
   346→        nodes: state.nodes.map((n) => ({ ...n })),
   347→        pods: state.pods.map((p) => ({ ...p })),
   348→        deployments: state.deployments.map((d) => ({ ...d })),
   349→        replicaSets: state.replicaSets.map((r) => ({ ...r })),
   350→        daemonSets: state.daemonSets.map((d) => ({ ...d })),
   351→        statefulSets: state.statefulSets.map((d) => ({ ...d })),
   352→        services: state.services.map((d) => ({ ...d })),
   353→        configMaps: state.configMaps.map((d) => ({ ...d })),
   354→        secrets: state.secrets.map((d) => ({ ...d })),
   355→        cronJobs: state.cronJobs.map((d) => ({ ...d })),
   356→        jobs: state.jobs.map((d) => ({ ...d })),
   357→        events: [...state.events],
   358→      }
   359→      s.tickCount = state.tickCount + 1
   360→      const t = s.tickCount
   361→      const workers = () => s.nodes.filter((n) => n.role === 'worker')
   362→
   363→      const ownedBy = (oid: string) => s.pods.filter((p) => p.ownerId === oid)
   364→      const activeOf = (oid: string) => ownedBy(oid).filter((p) => p.phase !== 'Terminating' && p.phase !== 'Succeeded')
   365→
   366→      // ---- bind pods that already have a node (DS requiredNode / manual placement) ----
   367→      for (const pod of s.pods) {
   368→        if (pod.phase === 'Pending' && (pod.nodeId || pod.requiredNode)) {
   369→          const target = pod.nodeId ?? pod.requiredNode
   370→          pod.nodeId = target
   371→          pod.phase = 'ContainerCreating'
   372→          pod.phaseTicks = 0
   373→          const host = s.nodes.find((n) => n.id === target)
   374→          logEvent(s, 'Normal', 'Scheduled', `${pod.name} assigned to ${host?.name ?? target}`, pod.name)
   375→          podLog(pod, t, `Assigned to ${host?.name ?? target}`)
   376→        }
   377→      }
   378→
   379→      // ---- scheduler for unbound Pending pods ----
   380→      for (const pod of s.pods) {
   381→        if (pod.phase !== 'Pending' || pod.nodeId) continue
   382→        const cands = workers().filter((n) => {
   383→          const u = nodeUsage(s.pods, n.id)
   384→          return u.cpu + pod.cpuRequest <= n.cpuCapacity && u.mem + pod.memRequest <= n.memCapacity
   385→        })
   386→        if (cands.length > 0) {
   387→          cands.sort((a, b) => {
   388→            const ua = nodeUsage(s.pods, a.id)
   389→            const ub = nodeUsage(s.pods, b.id)
   390→            return ua.cpu / a.cpuCapacity - ub.cpu / b.cpuCapacity
   391→          })
   392→          pod.nodeId = cands[0].id
   393→          pod.phase = 'ContainerCreating'
   394→          pod.phaseTicks = 0
   395→          logEvent(s, 'Normal', 'Scheduled', `Successfully assigned ${pod.name} to ${cands[0].name}`, pod.name)
   396→          podLog(pod, t, `Assigned to ${cands[0].name}`)
   397→        } else if (t % 4 === 0) {
   398→          logEvent(s, 'Warning', 'FailedScheduling', `0/${workers().length} nodes are available: insufficient cpu/mem for ${pod.name}`, pod.name)
   399→        }
   400→      }
   401→
   402→      // ---- pod lifecycle ----
   403→      for (const pod of s.pods) {
   404→        pod.ageTicks += 1
   405→        pod.phaseTicks += 1
   406→        switch (pod.phase) {
   407→          case 'ContainerCreating':
   408→            if (pod.phaseTicks >= 3) {
   409→              pod.phase = 'Running'
   410→              pod.phaseTicks = 0
   411→              logEvent(s, 'Normal', 'Started', `Started container ${pod.image} in pod ${pod.name}`, pod.name)
   412→              podLog(pod, t, `Container started: ${pod.image}`)
   413→            }
   414→            break
   415→          case 'Running':
   416→            if (pod.completionTicks !== null) {
   417→              pod.completionTicks -= 1
   418→              if (pod.completionTicks <= 0) {
   419→                pod.phase = 'Succeeded'
   420→                pod.phaseTicks = 0
   421→                logEvent(s, 'Normal', 'Completed', `Pod ${pod.name} completed (job)`, pod.name)
   422→                podLog(pod, t, 'Job completed successfully')
   423→              }
   424→            } else if (Math.random() < pod.crashRate) {
   425→              pod.phase = 'CrashLoopBackOff'
   426→              pod.phaseTicks = 0
   427→              pod.restarts += 1
   428→              logEvent(s, 'Warning', 'BackOff', `Back-off restarting failed container in ${pod.name} (restart #${pod.restarts})`, pod.name)
   429→              podLog(pod, t, 'ERROR: container exited (code 137)')
   430→            }
   431→            break
   432→          case 'CrashLoopBackOff': {
   433→            const backoff = Math.min(2 ** Math.min(pod.restarts, 4), 16)
   434→            if (pod.phaseTicks >= backoff) {
   435→              pod.phase = 'ContainerCreating'
   436→              pod.phaseTicks = 0
   437→              logEvent(s, 'Normal', 'Pulling', `Restarting container ${pod.image} in ${pod.name}`, pod.name)
   438→            }
   439→            break
   440→          }
   441→          case 'Terminating':
   442→            if (pod.phaseTicks >= 2) {
   443→              // pod will be cleaned up below
   444→            }
   445→            break
   446→          default:
   447→            break
   448→        }
   449→      }
   450→
   451→      // ---- cleanup finished pods ----
   452→      s.pods = s.pods.filter((p) => {
   453→        if (p.phase === 'Terminating' && p.phaseTicks >= 2) return false
   454→        if (p.phase === 'Succeeded' && p.phaseTicks >= 4) return false
   455→        return true
   456→      })
   457→
   458→      // ---- deployment controller ----
   459→      for (const dep of s.deployments) {
   460→        // ensure active RS with desired spec
   461→        const active = s.replicaSets.filter((r) => r.parentDeployment === dep.id && !r.superseded)
   462→        const matchesSpec = active.length > 0 && active[0].image === dep.image && active[0].replicas === dep.replicas
   463→        if (!matchesSpec && active.length > 0) {
   464→          // rollout: supersede old RS
   465→          for (const old of active) old.superseded = true
   466→          const rs: ReplicaSet = {
   467→            id: uid('rs'),
   468→            name: `${dep.name}-${rand(6)}`,
   469→            labels: { ...dep.labels },
   470→            selector: { ...dep.selector },
   471→            replicas: dep.replicas,
   472→            image: dep.image,
   473→            cpuRequest: dep.cpuRequest,
   474→            memRequest: dep.memRequest,
   475→            crashRate: dep.crashRate,
   476→            revision: dep.revision,
   477→            parentDeployment: dep.id,
   478→            superseded: false,
   479→          }
   480→          s.replicaSets.push(rs)
   481→          logEvent(s, 'Normal', 'RolloutStarted', `Deployment ${dep.name} rolling out revision ${dep.revision} (${dep.image})`, dep.name)
   482→        } else if (active.length === 0) {
   483→          const rs: ReplicaSet = {
   484→            id: uid('rs'),
   485→            name: `${dep.name}-${rand(6)}`,
   486→            labels: { ...dep.labels },
   487→            selector: { ...dep.selector },
   488→            replicas: dep.replicas,
   489→            image: dep.image,
   490→            cpuRequest: dep.cpuRequest,
   491→            memRequest: dep.memRequest,
   492→            crashRate: dep.crashRate,
   493→            revision: dep.revision,
   494→            parentDeployment: dep.id,
   495→            superseded: false,
   496→          }
   497→          s.replicaSets.push(rs)
   498→          logEvent(s, 'Normal', 'SuccessfulCreate', `Created ReplicaSet ${rs.name} with ${rs.replicas} replica(s)`, dep.name)
   499→        }
   500→
   501→        // superseded RS scale-down: one pod at a time when new RS has capacity to schedule
   502→        const newest = s.replicaSets.find((r) => r.parentDeployment === dep.id && !r.superseded)
   503→        if (newest) {
   504→          for (const old of s.replicaSets) {
   505→            if (!old.superseded || old.parentDeployment !== dep.id) continue
   506→            const oldPods = ownedBy(old.id).filter((p) => p.phase !== 'Terminating' && p.phase !== 'Succeeded')
   507→            const newPods = activeOf(newest.id)
   508→            // scale down one old pod per tick once new pods are up (rolling)
   509→            if (oldPods.length > 0 && newPods.filter((p) => p.phase === 'Running').length >= 1) {
   510→              const victim = oldPods[0]
   511→              victim.phase = 'Terminating'
   512→              victim.phaseTicks = 0
   513→              logEvent(s, 'Normal', 'Killing', `Stopping container ${victim.image} in pod ${victim.name} (rollout)`, dep.name)
   514→              old.replicas = Math.max(0, old.replicas - 1)
   515→            } else if (oldPods.length === 0 && old.replicas > 0) {
   516→              old.replicas = 0
   517→            }
   518→          }
   519→        }
   520→      }
   521→
   522→      // ---- replicaset controller ----
   523→      for (const rs of s.replicaSets) {
   524→        const pods = ownedBy(rs.id).filter((p) => p.phase !== 'Terminating' && p.phase !== 'Succeeded')
   525→        if (pods.length < rs.replicas) {
   526→          const need = rs.replicas - pods.length
   527→          for (let i = 0; i < need; i++) {
   528→            s.pods.push(
   529→              basePod({
   530→                name: `${rs.name}-${rand(5)}-${rand(5)}`,
   531→                namespace: rs.namespace,
   532→                labels: { ...rs.labels, pod: rand(4) },
   533→                ownerId: rs.id,
   534→                ownerKind: 'replicaset',
   535→                image: rs.image,
   536→                cpuRequest: rs.cpuRequest,
   537→                memRequest: rs.memRequest,
   538→                crashRate: rs.crashRate,
   539→              }) as never
   540→            )
   541→            logEvent(s, 'Normal', 'SuccessfulCreate', `Created pod for ReplicaSet ${rs.name}`, rs.name)
   542→          }
   543→        } else if (pods.length > rs.replicas) {
   544→          for (const victim of pods.slice(rs.replicas)) {
   545→            victim.phase = 'Terminating'
   546→            victim.phaseTicks = 0
   547→            logEvent(s, 'Normal', 'Killing', `Stopping container ${victim.image} in pod ${victim.name}`, rs.name)
   548→          }
   549→        }
   550→      }
   551→
   552→      // ---- daemonset controller ----
   553→      for (const ds of s.daemonSets) {
   554→        for (const n of workers()) {
   555→          const existing = s.pods.filter((p) => p.ownerId === ds.id && p.requiredNode === n.id && p.phase !== 'Terminating' && p.phase !== 'Succeeded')
   556→          if (existing.length === 0) {
   557→            s.pods.push(
   558→              basePod({
   559→                name: `${ds.name}-${rand(5)}-${rand(5)}`,
   560→                namespace: ds.namespace,
   561→                labels: { ...ds.labels, pod: rand(4) },
   562→                ownerId: ds.id,
   563→                ownerKind: 'daemonset',
   564→                image: ds.image,
   565→                cpuRequest: ds.cpuRequest,
   566→                memRequest: ds.memRequest,
   567→                crashRate: ds.crashRate,
   568→                requiredNode: n.id,
   569→              }) as never
   570→            )
   571→            logEvent(s, 'Normal', 'SuccessfulCreate', `DaemonSet ${ds.name} started pod on ${n.name}`, ds.name)
   572→          }
   573→        }
   574→      }
   575→
   576→      // ---- statefulset controller (ordered) ----
   577→      for (const sts of s.statefulSets) {
   578→        for (let i = 0; i < sts.replicas; i++) {
   579→          const name = `${sts.name}-${i}`
   580→          const existing = s.pods.find((p) => p.name === name && p.phase !== 'Terminating' && p.phase !== 'Succeeded')
   581→          if (!existing) {
   582→            // wait until all lower ordinals are Running
   583→            const lower: number[] = []
   584→            for (let j = 0; j < i; j++) {
   585→              const lp = s.pods.find((p) => p.name === `${sts.name}-${j}`)
   586→              if (!lp || lp.phase !== 'Running') lower.push(j)
   587→            }
   588→            if (lower.length > 0) continue
   589→            s.pods.push(
   590→              basePod({
   591→                name,
   592→                namespace: sts.namespace,
   593→                labels: { ...sts.labels, 'apps.kubernetes.io/pod-index': String(i), pod: name },
   594→                ownerId: sts.id,
   595→                ownerKind: 'statefulset',
   596→                image: sts.image,
   597→                cpuRequest: sts.cpuRequest,
   598→                memRequest: sts.memRequest,
   599→                crashRate: sts.crashRate,
   600→              }) as never
   601→            )
   602→            logEvent(s, 'Normal', 'SuccessfulCreate', `StatefulSet ${sts.name} created ${name} (ordinal ${i})`, sts.name)
   603→          }
   604→        }
   605→        // scale down: remove highest ordinal
   606→        const pods = s.pods.filter((p) => p.ownerId === sts.id && p.phase !== 'Terminating' && p.phase !== 'Succeeded')
   607→        if (pods.length > sts.replicas) {
   608→          const sorted = pods.sort((a, b) => (parseInt(a.name.split('-').pop() ?? '0') - parseInt(b.name.split('-').pop() ?? '0')))
   609→          const victim = sorted[sorted.length - 1]
   610→          victim.phase = 'Terminating'
   611→          victim.phaseTicks = 0
   612→          logEvent(s, 'Normal', 'Killing', `StatefulSet scaling down: stopping ${victim.name}`, sts.name)
   613→        }
   614→      }
   615→
   616→      // ---- cronjob controller ----
   617→      for (const cj of s.cronJobs) {
   618→        const interval = parseSchedule(cj.schedule)
   619→        if (interval <= 0) continue
   620→        if (t - cj.lastRun >= interval) {
   621→          cj.lastRun = t
   622→          const job: Job = {
   623→            id: uid('job'),
   624→            name: `${cj.name}-${t}`,
   625→            namespace: cj.namespace,
   626→            image: cj.image,
   627→            duration: cj.duration,
   628→            status: 'Active',
   629→            cronjobOwner: cj.id,
   630→            createdTick: t,
   631→          }
   632→          s.jobs.push(job)
   633→          s.pods.push(
   634→            basePod({
   635→              name: `${job.name}-${rand(5)}`,
   636→              namespace: cj.namespace,
   637→              labels: { 'k8s kind': 'job', job: job.name },
   638→              ownerId: job.id,
   639→              ownerKind: 'job',
   640→              image: cj.image,
   641→              cpuRequest: cj.cpuRequest,
   642→              memRequest: cj.memRequest,
   643→              crashRate: 0,
   644→              completionTicks: cj.duration,
   645→            }) as never
   646→          )
   647→          cj.active += 1
   648→          logEvent(s, 'Normal', 'SuccessfulCreate', `CronJob ${cj.name} spawned Job ${job.name}`, cj.name)
   649→        }
   650→      }
   651→
   652→      // ---- job status tracking ----
   653→      for (const job of s.jobs) {
   654→        if (job.status === 'Active') {
   655→          const pods = s.pods.filter((p) => p.ownerId === job.id)
   656→          const done = pods.length > 0 && pods.every((p) => p.phase === 'Succeeded') && pods.some((p) => p.phaseTicks >= 1)
   657→          if (done) {
   658→            job.status = 'Complete'
   659→            if (job.cronjobOwner) {
   660→              const cj = s.cronJobs.find((c) => c.id === job.cronjobOwner)
   661→              if (cj) {
   662→                cj.active -= 1
   663→                cj.completions += 1
   664→              }
   665→            }
   666→            logEvent(s, 'Normal', 'Completed', `Job ${job.name} completed`, job.name)
   667→          } else if (t - job.createdTick > 600) {
   668→            job.status = 'Complete' // safety
   669→          }
   670→        }
   671→      }
   672→      // GC completed jobs older than 30 ticks
   673→      s.jobs = s.jobs.filter((j) => !(j.status === 'Complete' && t - j.createdTick > 30))
   674→
   675→      // ---- service LB ip assignment ----
   676→      for (const svc of s.services) {
   677→        if (svc.type === 'LoadBalancer' && !svc.externalIP && svc.lbTicks > 0) {
   678→          svc.lbTicks -= 1
   679→          if (svc.lbTicks === 0) {
   680→            svc.externalIP = `203.0.113.${10 + (s.services.length % 240)}`
   681→            logEvent(s, 'Normal', 'Assigned', `LoadBalancer ${svc.name} got external IP ${svc.externalIP}`, svc.name)
   682→          }
   683→        }
   684→      }
   685→
   686→      // ---- chaos monkey ----
   687→      if (s.chaos && t % 3 === 0) {
   688→        const alive = s.pods.filter((p) => p.phase === 'Running')
   689→        if (alive.length > 0 && Math.random() < 0.5) {
   690→          const victim = alive[Math.floor(Math.random() * alive.length)]
   691→          victim.phase = 'CrashLoopBackOff'
   692→          victim.phaseTicks = 0
   693→          victim.restarts += 1
   694→          logEvent(s, 'Warning', 'ChaosMonkey', `Random failure injected in pod ${victim.name} 🐒`, victim.name)
   695→        }
   696→      }
   697→
   698→      return s
   699→    }),

   945→          s.services = state.services.filter((v) => v.id !== id)

   979→          s.configMaps = state.configMaps.filter((c) => c.namespace !== ns.name)
   980→          s.secrets = state.secrets.filter((c) => c.namespace !== ns.name)
